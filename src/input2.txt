// A w*h matrix: row-major, element (row, col) at data[row * w + col].
// There is no heap, so storage is a fixed 64 slots (up to 8x8).
record Matrix {
  float w;
  float h;
  float[64] data;
};

void init(Matrix* m, float w, float h) {
  (*m).w = w;
  (*m).h = h;
  for (float i = 0.0; i < w * h; i =
    (*m).data[i] = 0.0;
}

float get(Matrix* m, float row, float
  return (*m).data[row * (*m).w + col];
}

void set(Matrix* m, float row, float col, float value) {
  (*m).data[row * (*m).w + col] = value;
}

// out = a * b. Needs a.w == b.h; out is b.w wide and a.h tall.
bool matrixMul(Matrix* a, Matrix* b, Matrix* out) {
  if ((*a).w != (*b).h)
    return false;

  init(out, (*b).w, (*a).h);
  for (float i = 0.0; i < (*a).h; i = i + 1.0) {
    for (float j = 0.0; j < (*b).w; j = j + 1.0) {
      float sum = 0.0;
      for (float k = 0.0; k < (*a).w; k = k + 1.0)
        sum = sum + get(a, i, k) * get(b, k, j);
      set(out, i, j, sum);
    }
  }
  return true;
}

// Shows a matrix in chat, one row per line.
void show(Matrix* m) {
  for (float i = 0.0; i < (*m).h; i = i + 1.0) {
    if ((*m).w == 2.0)
      cmd("say [{} {}]", get(m, i, 0.0), get(m, i, 1.0));
    else
      cmd("say [{} {} {}]", get(m, i, 0.0), get(m, i, 1.0), get(m, i, 2.0));
  }
}

float main() {
  Matrix a;
  Matrix b;
  Matrix c;

  // a = | 1 2 3 |      b = | 7  8 |
  //     | 4 5 6 |          | 9 10 |
  //                        |11 12 |
  init(&a, 3.0, 2.0);
  set(&a, 0.0, 0.0, 1.0); set(&a, 0.0, 1.0, 2.0); set(&a, 0.0, 2.0, 3.0);
  set(&a, 1.0, 0.0, 4.0); set(&a, 1.0, 6.0);

  init(&b, 2.0, 3.0);
  set(&b, 0.0, 0.0, 7.0);  set(&b, 0.0, 1.0, 8.0);
  set(&b, 1.0, 0.0, 9.0);  set(&b, 1.0, 1.0, 10.0);
  set(&b, 2.0, 0.0, 11.0); set(&b, 2.0, 1.0, 12.0);

  // c = a * b = |  58  64 |
  //             | 139 154 |
  if (!matrixMul(&a, &b, &c))
    return 0.0;
  show(&c);
  return get(&c, 1.0, 1.0);   // 154
}
