// Exercises the whole language: records, member access, the operator
// cascade, control flow and bool literals.

record Point {
  float x;
  float y;
};

record Segment {  // two endpoints and a flag
  Point head;
  Point tail;
  bool closed;
};

float test(Segment s1, float v)
{
return s1.head.x + v;
}

void add(&float v)
{
  v = v + 1.0;
}

float main() {
    Segment seg;
    float a = 3.0;
    add(&a);
    seg.head.x = test(seg, 3.5) + a;
    return seg.head.x;
}
