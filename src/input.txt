// Exercises the whole language: records, member access, the operator
// cascade, control flow and bool literals.

record Point {
  float x;
  float y;
};

record Segment {  // two endpoints and a flag
  Point head;
  Point tail;
  bool closed;
};

float test(Segment s1)
{
return s1.head.x + 1.0;
}

void main() {
    Segment seg;
    seg.head.x = test(seg);
}
