void grid(float w, float h)
{
  for(float i = 0.0; i < h; i=i+1.0)
  {
    for(float j = 0.0; j < w; j=j+1.0)
    {
      cmd("setblock {} 100 {} diamond_block", j, i);
    }
  }
}

// A w*h matrix as a float**: `rows` points at h row pointers, and each row
// pointer points at w floats. m.rows[i][j] is row i, column j, like in C.
//
// There is no heap, so the caller owns the storage: a float[] for the numbers
// and a float*[] for the row pointers. init() wires the rows into it.
record Matrix {
  float w;
  float h;
  float** rows;
};

void init(Matrix* m, float w, float h, float** rowSlots, float* data) {
  (*m).w = w;
  (*m).h = h;
  (*m).rows = rowSlots;
  for (float i = 0.0; i < h; i = i + 1.0) {
    rowSlots[i] = &data[i * w];          // row i starts i*w numbers in
    for (float j = 0.0; j < w; j = j + 1.0)
      rowSlots[i][j] = 0.0;
  }
}

// out = a * b. Needs a.w == b.h; out must already be b.w wide and a.h tall.
bool matrixMul(Matrix* a, Matrix* b, Matrix* out) {
  if ((*a).w != (*b).h || (*out).w != (*b).w || (*out).h != (*a).h)
    return false;

  float** x = (*a).rows;
  float** y = (*b).rows;
  float** z = (*out).rows;
  for (float i = 0.0; i < (*a).h; i = i + 1.0) {
    float* row = z[i];
    for (float j = 0.0; j < (*b).w; j = j + 1.0) {
      float sum = 0.0;
      for (float k = 0.0; k < (*a).w; k = k + 1.0)
        sum = sum + x[i][k] * y[k][j];
      row[j] = sum;
    }
  }
  return true;
}

// Shows a matrix in chat, one row per line.
void show(Matrix* m) {
  for (float i = 0.0; i < (*m).h; i = i + 1.0) {
    float* row = (*m).rows[i];
    if ((*m).w == 2.0)
      cmd("say [{} {}]", row[0.0], row[1.0]);
    else
      cmd("say [{} {} {}]", row[0.0], row[1.0], row[2.0]);
  }
}

float main() {
  // Storage for three matrices of up to 8x8.
  float[64] aData;  float*[8] aRows;
  float[64] bData;  float*[8] bRows;
  float[64] cData;  float*[8] cRows;

  Matrix a;
  Matrix b;
  Matrix c;
  init(&a, 3.0, 2.0, &aRows[0.0], &aData[0.0]);
  init(&b, 2.0, 3.0, &bRows[0.0], &bData[0.0]);
  init(&c, 2.0, 2.0, &cRows[0.0], &cData[0.0]);

  // a = | 1 2 3 |      b = | 7  8 |
  //     | 4 5 6 |          | 9 10 |
  //                        |11 12 |
  float** ar = a.rows;
  ar[0.0][0.0] = 1.0; ar[0.0][1.0] = 2.0; ar[0.0][2.0] = 3.0;
  ar[1.0][0.0] = 4.0; ar[1.0][1.0] = 5.0; ar[1.0][2.0] = 6.0;

  b.rows[0.0][0.0] = 7.0;  b.rows[0.0][1.0] = 8.0;
  b.rows[1.0][0.0] = 9.0;  b.rows[1.0][1.0] = 10.0;
  b.rows[2.0][0.0] = 11.0; b.rows[2.0][1.0] = 12.0;

  // c = a * b = |  58  64 |
  //             | 139 154 |
  if (!matrixMul(&a, &b, &c))
    return 0.0;
  show(&c);

  // The rows really are pointers into cData: row 1 starts at cData[2].
  return c.rows[1.0][1.0] + cData[2.0];   // 154 + 139 = 293
}
