record Point {
  float x;
  float y;
};

record Segment {
  Point head;
  Point tail;
  bool closed;
};

float square(float x) {
  return x * x;
}

float lenSq(Segment s) {
  float dx = s.tail.x - s.head.x;
  float dy = s.tail.y - s.head.y;
  return square(dx) + square(dy);
}

float clamp(float v, float lo, float hi) {
  if (v < lo) {
    return lo;
  } else if (v > hi) {
    return hi;
  } else {
    return v;
  }
}

float walk(Segment s, float steps) {
  float total = 0;
  bool active = !s.closed || true;
  for (float i = 0; i < steps; i = i + 1) {
    if (!(i == 0) && s.closed == true && active) {
      total = total + lenSq(s) / i;
    } else {
      total = total + 1;
    }
  }
  while (total > 100 || total != total) {
    total = total / 2;
  }
  return clamp(total, 0, 100);
}

float poly(float a, float b, float c, float t) {
  float tsq = t * t;
  float acc = a * tsq + b * t + c;
  acc = acc / (1 + tsq) - 0.5;
  return acc;
}
